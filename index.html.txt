<script>
let xp = Number(localStorage.getItem("xp")) || 0
let rubies = Number(localStorage.getItem("rubies")) || 0
let lives = Number(localStorage.getItem("lives")) || 5
let friends = JSON.parse(localStorage.getItem("friends")) || []
let messages = JSON.parse(localStorage.getItem("messages")) || []

update()
displayFriends()
displayMessages()

function signup(){
let u=document.getElementById("user").value
let p=document.getElementById("pass").value
localStorage.setItem("user",u)
localStorage.setItem("pass",p)
alert("Account created!")
}

function login(){
let u=document.getElementById("loginUser").value
let p=document.getElementById("loginPass").value
if(u==localStorage.getItem("user") && p==localStorage.getItem("pass")){
document.getElementById("auth").style.display="none"
document.getElementById("dashboard").style.display="block"
document.getElementById("player").innerText=u
updateLeaderboard()
}else{
alert("Wrong login")
}
}

function lesson(lang){
if(lives<=0){ alert("No lives left!"); return }
let ans=prompt("Theo asks: write a keyword used in "+lang)
if(ans){ xp+=25; alert("Nice coding!") } 
else{ lives--; alert("Try again!") }
save(); unlockFriends(); update()
}

function bugGame(){
let g=prompt("Bug Hunter: which symbol ends a JS line?")
if(g==";"){ xp+=40; alert("Bug fixed!") } 
else{ lives--; alert("Bug still there!") }
save(); update()
}

function speedGame(){
let g=prompt("Speed Code: what keyword creates a function in JS?")
if(g=="function"){ xp+=60; alert("Fast coder!") } 
else{ lives-- }
save(); update()
}

function daily(){
xp+=80; alert("Daily mission complete!")
save(); update()
}

function theoHelp(){
let tips=["Loops repeat instructions.","Functions organize your code.","Debugging teaches you a lot.","Break big problems into smaller steps.","Practice coding daily."]
let r=Math.floor(Math.random()*tips.length)
document.getElementById("theoText").innerText=tips[r]
}

function unlockFriends(){
if(xp>=300) rubies=2
if(xp>=700) rubies=5
if(xp>=1200) rubies=10
if(rubies>=2) document.getElementById("friends").innerText="Byte 🤖"
if(rubies>=5) document.getElementById("friends").innerText="Byte 🤖 + Pyra 🐍"
if(rubies>=10) document.getElementById("friends").innerText="Byte 🤖 + Pyra 🐍 + Volt ⚡"
}

function update(){
document.getElementById("xp").innerText=xp
document.getElementById("rubies").innerText=rubies
document.getElementById("lives").innerText=lives
}

function save(){
localStorage.setItem("xp",xp)
localStorage.setItem("rubies",rubies)
localStorage.setItem("lives",lives)
}

function updateLeaderboard(){
let name=localStorage.getItem("user")
document.getElementById("leaderboard").innerText=name+" — "+xp+" XP"
}

function addFriend(){
let name=document.getElementById("friendName").value
if(name){ friends.push(name)
localStorage.setItem("friends",JSON.stringify(friends))
displayFriends()
}
}

function displayFriends(){
let list=document.getElementById("friendList")
list.innerHTML=""
friends.forEach(f=>{
let li=document.createElement("li")
li.innerText=f
list.appendChild(li)
})
}

function sendMessage(){
let msg=document.getElementById("chatInput").value
if(msg){ messages.push(msg)
localStorage.setItem("messages",JSON.stringify(messages))
displayMessages()
}
}

function displayMessages(){
let chat=document.getElementById("chatWindow")
chat.innerHTML=""
messages.forEach(m=>{
let p=document.createElement("p")
p.innerText=m
chat.appendChild(p)
})
}
